## 三、参数配置说明 ⚙️

### 3.1 核心参数

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `k` | int | 3 | 聚类数量（建议2-5） |
| `features` | list | ["recency_days", "frequency", "monetary"] | 聚类特征 |
| `use_optimized` | bool | true | 是否使用优化版 |
| `use_minibatch` | bool | true | 是否启用MiniBatch（数据>5万时自动启用） |
| `use_cache` | bool | true | 是否启用缓存 |

### 3.2 高级参数（可选）

```json
{
  "parameters": {
    "k": 3,
    "features": ["recency_days", "frequency", "monetary"],
    "use_optimized": true,

    // 高级配置
    "minibatch_config": {
      "batch_size": 10000,      // MiniBatch批大小
      "n_init": 10,             // 初始化次数
      "max_iter": 300           // 最大迭代次数
    },
    "cache_config": {
      "ttl": 3600,              // 缓存过期时间（秒）
      "force_refresh": false    // 强制刷新缓存
    }
  }
}
```

```bash
rm cache/models/a3f2b8c1d4e5f6g7_models.pkl
```

#### 清除过期缓存（Python脚本）
```python
import os
import time
from pathlib import Path

def clean_expired_cache(cache_dir, ttl=3600):
    """清除超过ttl秒的缓存文件"""
    now = time.time()
    for file in Path(cache_dir).glob("*"):
        if now - file.stat().st_mtime > ttl:
            file.unlink()
            print(f"已删除过期缓存: {file}")

# 执行清理
clean_expired_cache("cache/clustering", ttl=3600)  # 1小时
clean_expired_cache("cache/models", ttl=86400)     # 24小时
```

### 7.1 A/B对比测试

#### 测试1：原版 vs 优化版性能对比
```python
import time
import requests

def test_clustering(use_optimized):
    start = time.time()

    # 创建任务
    response = requests.post("http://localhost:8000/api/v1/analysis/tasks",
        json={
            "task_name": f"测试{'优化版' if use_optimized else '原版'}",
            "task_type": "clustering",
            "parameters": {
                "k": 3,
                "features": ["recency_days", "frequency", "monetary"],
                "use_optimized": use_optimized,
                "use_cache": False  # 禁用缓存，测试真实性能
            }
        },
        headers={"Authorization": "Bearer YOUR_TOKEN"}
    )

    task_id = response.json()["task_id"]

    # 轮询直到完成
    while True:
        status = requests.get(f"http://localhost:8000/api/v1/analysis/tasks/{task_id}",
            headers={"Authorization": "Bearer YOUR_TOKEN"}
        ).json()

        if status["status"] == "completed":
            break
        time.sleep(2)

    elapsed = time.time() - start
    print(f"{'优化版' if use_optimized else '原版'} 耗时: {elapsed:.2f}秒")

    return elapsed

# 执行对比测试
original_time = test_clustering(use_optimized=False)
optimized_time = test_clustering(use_optimized=True)

print(f"\n性能提升: {(original_time / optimized_time - 1) * 100:.1f}%")
```

### 9.3 结果验证

```python
def validate_clustering_result(result):
    """验证聚类结果质量"""
    checks = []

    # 检查1: Silhouette Score应 > 0.3
    checks.append(result['silhouette_score'] > 0.3)

    # 检查2: Davies-Bouldin Index应 < 2.0
    checks.append(result['davies_bouldin'] < 2.0)

    # 检查3: 每个簇至少有5%的数据
    for cluster_stat in result['kpi_summary'].values():
        checks.append(cluster_stat['percentage'] >= 5.0)

    # 检查4: 簇标签不应重复
    labels = [v for k, v in result['cluster_labels'].items()]
    checks.append(len(labels) == len(set(labels)))

    if all(checks):
        print("✅ 聚类结果验证通过")
        return True
    else:
        print("❌ 聚类结果验证失败")
        return False
```
