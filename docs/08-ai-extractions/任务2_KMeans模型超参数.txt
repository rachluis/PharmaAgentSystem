# m, f, r 为群组均值；gm, gf, gr 为全局均值
if m > gm * 2 and f > gf * 1.5:
    return "High-Value Core (VIP)"    # 货币贡献超全局均值200%且频次超全局均值150%

    return "Growth Potential"         # 货币贡献超150%且频次超均值
